## 매개변수 방정식 기반 도형을 그리고 변형하기

다음의 식을 가지는 하트 방정식을 평면상에 그리고
스케일, 회전, 이동 시킨다.
$$\begin{aligned}
&x=16\sin^3\theta\\ 
&y=13cos\theta - 5cos2\theta - 2\cos3\theta - \cos4\theta\\ 
&(0\leq\theta\leq2\pi)
\end{aligned}$$

```cpp
// 하트의 위치, 크기, 회전값 변수
Vector2 currentPosition;
float currentScale = 10.f;
float currentDegree = 0.f;

void SoftRenderer::Update2D(float InDeltaSeconds)
{
	auto& g = Get2DGameEngine();
	const InputManager& input = g.GetInputManager();

	static float moveSpeed = 100.f;
	static float scaleSpeed = 20.f;
	static float scaleMin = 5.f;
	static float scaleMax = 20.f;
	static float duration = 1.5f;
	static float elapsedTime = 0.f;
	static float rotateSpeed = 180.f;

	Vector2 inputVector = Vector2(input.GetAxis(InputAxis::XAxis), input.GetAxis(InputAxis::YAxis)).GetNormalize();
	Vector2 deltaPosition = inputVector * moveSpeed * InDeltaSeconds;
	float deltaDegree = input.GetAxis(InputAxis::WAxis) * rotateSpeed * InDeltaSeconds;

	// 경과 시간과 사인함수를 활용한 [0,1]값의 생성
	elapsedTime += InDeltaSeconds; // 시간 누적
	elapsedTime = Math::FMod(elapsedTime, duration); // duration 만큼의 시간이 지나면 0으로 초기화
	float currentRad = (elapsedTime / duration) * Math::TwoPI; // 1주기(2pi) = 1duration
	float alpha = (sinf(currentRad) + 1) * 0.5f; // [0,1]범위로 스케일된 사인함수에 값 대입


	// 물체의 최종 상태 설정
	currentPosition += deltaPosition;
	currentScale = Math::Lerp(scaleMin, scaleMax, alpha); // min~max 사이에서 값 결정
	currentDegree += deltaDegree;
}

void SoftRenderer::Render2D()
{
	auto& r = GetRenderer();
	const auto& g = Get2DGameEngine();
	DrawGizmo2D();

	float rad = 0.f;
	static float increment = 0.001f;
	static std::vector<Vector2> hearts;
	HSVColor hsv(0.f, 1.f, 0.85f);

	// 하트를 구성하는 점 생성
	if (hearts.empty())
	{
		for (rad = 0.f; rad < Math::TwoPI; rad += increment)
		{
			float sin = sinf(rad);
			float cos = cosf(rad);
			float cos2 = cosf(2 * rad);
			float cos3 = cosf(3 * rad);
			float cos4 = cosf(4 * rad);
			float x = 16.f * sin * sin * sin;
			float y = 13 * cos - 5 * cos2 - 2 * cos3 - cos4;
			hearts.push_back(Vector2(x, y));
		}
	}

	// 각도에 해당하는 사인과 코사인 값 얻기 (회전 변환용 기저벡터의 요소)
	float sin = 0.f, cos = 0.f;
	Math::GetSinCos(sin, cos, currentDegree);

	// 출력
	rad = 0.f;
	for (auto const& v : hearts)
	{
		Vector2 scaledV = v * currentScale; // 크기 적용
		Vector2 rotatedV = Vector2(scaledV.X * cos - scaledV.Y * sin, scaledV.X * sin + scaledV.Y * cos); // 회전 적용 (선형변환)
		Vector2 translatedV = rotatedV + currentPosition; // 이동

		hsv.H = rad / Math::TwoPI; // 2pi = 1 에 대응
		r.DrawPoint(translatedV, hsv.ToLinearColor());
		rad += increment;
	}

	r.PushStatisticText(std::string("Position : ") + currentPosition.ToString());
	r.PushStatisticText(std::string("Scale : ") + std::to_string(currentScale));
	r.PushStatisticText(std::string("Degree : ") + std::to_string(currentDegree));
}
```

## 극좌표계

아크탄젠트2(atan2)
일반 아크탄젠트 함수는 1, 4사분면 내 벡터의 사잇각은 잘 구해준다.
그러나 2, 3사분면 내의 벡터의 경우, 탄젠트 값이 1, 4사분면 위의 것과 다를 바가 없기에
언제나 결과는 $(-90\degree,90\degree)$에 속하게 된다.

탄젠트 값 $y/x$을 받는 아크탄젠트 함수와 달리, 아트탄젠트2 함수는 벡터의 $x$와 $y$값을 각각 받아
각자의 부호를 통해 벡터가 몇 사분면 위에 있는지에 대한 정보를 살린다.
```
atan(float tan)
atan2(float y, float x)
```
해당 벡터가 2, 3사분면 위에 있을 경우 기존 아크탄젠트 값에 
$\pi$를 더하거나 빼서 $(-180\degree,180\degree]$내의 정확한 사잇각을 도출한다.

원점에서의 거리 및 회전 각도를 통해 점의 위치를 특정하는 좌표계를 극좌표계(Plar coordinate system)라고 한다.
데카르트 좌표계의 점을 극좌표계로 변환할 때, 사잇각의 크기를 구하기 위해 $\operatorname{atan2}$를 사용한다.
$$\begin{aligned}
&(r,\theta) = (\sqrt{x^2+y^2},\operatorname{atan2}(y,x)) \\
&(x,y)=(r\cos{\theta},r\sin{\theta})
\end{aligned}$$

### 극좌표계를 이용한 회전
일반적으로 기존 데카트르 좌표를 극좌표로 변환하여 회전을 간단하게 처리 후, 다시 데카르트 좌표계로 되돌려 출력한다.
아래 코드는 입력된 회전값에 따라 사각형을 원점에서 멀 수록 더 많이 회전시켜 일그러지게 만든다.

```cpp
float currentDegree = 0.f;
void SoftRenderer::Update2D(float InDeltaSeconds)
{
	auto& g = Get2DGameEngine();
	const InputManager& input = g.GetInputManager();
	static float rotateSpeed = 10.f;
	float deltaDegree = input.GetAxis(InputAxis::WAxis) * rotateSpeed * InDeltaSeconds;
	currentDegree += deltaDegree;
}

void SoftRenderer::Render2D()
{
	auto& r = GetRenderer();
	const auto& g = Get2DGameEngine();
	DrawGizmo2D();

	// 사각형 한 변의 절반 크기
	static float halfSize = 100.f;
	static std::vector<Vector2> squares;

	// 사각형을 구성하는 점을 생성
	if (squares.empty())
	{
		for (float x = -halfSize; x <= halfSize; x += 0.25f)
		{
			for (float y = -halfSize; y <= halfSize; y += 0.25f)
			{
				squares.push_back(Vector2(x, y));
			}
		}
	}

	float sin = 0.f, cos = 0.f;
	Math::GetSinCos(sin, cos, currentDegree);

	// 화면의 크기로부터 길이를 비교할 기준량 결정
	static float maxLength = Vector2(_ScreenSize.X, _ScreenSize.Y).Size() * 0.5f;

	// 원을 구성하는 점을 그린다.
	HSVColor hsv(0.f, 1.f, 0.85f);
	for (auto const& v : squares)
	{
		// 극좌표계로 변경한다.
		// (X:윈점에서의 거리, Y:각도)
		Vector2 polarV = v.ToPolarCoordinate();

		if (polarV.Y < 0.f)
		{
			// atan2에 의해 (-pi~pi)범위에 놓인 각도를 (0,2pi)로 변경
			polarV.Y += Math::TwoPI;
		}
		// (0,2pi) -> (0,1) -> 색상
		hsv.H = polarV.Y / Math::TwoPI;

		// 기준량에 가까울수록 회전값 만큼만 돌리고, 멀 수록 최대 5배 더 돌린다.
		float ratio = polarV.X / maxLength;
		float weight = Math::Lerp(1.f, 5.f, ratio);
		polarV.Y += Math::Deg2Rad(currentDegree) * weight;

		// 다시 데카르트 좌표계 변환 후 출력
		Vector2 cartesianV = polarV.ToCartesianCoordinate();
		r.DrawPoint(cartesianV, hsv.ToLinearColor());
	}
	r.PushStatisticText(std::string("Degree : ") + std::to_string(currentDegree));
}
```
